```javascript
const [A, B] = require("fs")
  .readFileSync("/dev/stdin")
  .toString()
  .split(" ")
  .map((el) => el.trim());

const minLen = A.length;
const maxLen = B.length;
const sets = new Set();

const dfs = (dep, nowWord) => {
  if (dep > maxLen) return;
  if (+nowWord >= +A && +nowWord <= +B) {
    sets.add(nowWord);
  }
  dfs(dep + 1, nowWord + "4");
  dfs(dep + 1, nowWord + "7");
};

dfs(0, "");
console.log(sets.size);
```
