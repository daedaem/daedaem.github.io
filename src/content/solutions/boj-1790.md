```javascript
(function solve() {
  const [N, K] = require("fs")
    .readFileSync(0)
    .toString()
    .trim()
    .split(" ")
    .map(Number);

  let length = 0;
  for (let i = 1; i <= N; i++) {
    length += i.toString().length;
    if (length >= K) {
      const numStr = i.toString();
      const idx = K - (length - numStr.length);
      console.log(numStr[idx - 1]);
      break;
    }
  }

  if (length < K) {
    console.log("-1");
  }
})();
```

> **정리(2026-09-06):** 실행 폴더에 따라 달라지는 상대 경로 대신 표준입력 파일 디스크립터 `0`을 읽도록 바꿨다. 풀이 로직은 바꾸지 않았다.
