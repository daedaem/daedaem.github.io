def dfs(start, end):
    visited[start] = True
    if False not in visited:
        global cnt
        return cnt
    else:
        for i in tree[start]:
            if visited[i] == True:
                pass
            else:
                cnt+=1
                visited[i] = True
                dfs(i, end)
